
######## unseen_data (coupon checkout) ########
-- unseen_data / coupon_field_shown
CREATE TABLE atlys.coupon_field_shown
(
    `os` String,
    `user_id` String,
    `currency` String,
    `timestamp` DateTime64(3),
    `cart_value` String,
    `destination` String,
    `device_type` String,
    `application_id` String,
    `geoip_country_code` String
)
ENGINE = SharedMergeTree('/clickhouse/tables/{uuid}/{shard}', '{replica}')
PARTITION BY toYYYYMM(timestamp)
ORDER BY (toDate(timestamp), device_type, destination, user_id, application_id)
SETTINGS index_granularity = 8192
-- MV for coupon_field_shown
CREATE MATERIALIZED VIEW atlys.mv_coupon_field_shown_to_activity TO atlys.activity_events
(
    `event_name` String,
    `ch_table` String,
    `timestamp` DateTime64(3),
    `user_id` String,
    `application_id` String,
    `device_type` String,
    `os` String,
    `geoip_country_code` String,
    `destination` String,
    `payload` String
)
AS SELECT
    'coupon_field_shown' AS event_name,
    'coupon_field_shown' AS ch_table,
    timestamp AS timestamp,
    toString(user_id) AS user_id,
    toString(application_id) AS application_id,
    toString(device_type) AS device_type,
    toString(os) AS os,
    toString(geoip_country_code) AS geoip_country_code,
    toString(destination) AS destination,
    toJSONString(mapFromArrays(['os', 'user_id', 'currency', 'timestamp', 'cart_value', 'destination', 'device_type', 'application_id', 'geoip_country_code'], [toString(os), toString(user_id), toString(currency), toString(timestamp), toString(cart_value), toString(destination), toString(device_type), toString(application_id), toString(geoip_country_code)])) AS payload
FROM atlys.coupon_field_shown
